SQL used (no ids), for the "after" re-run:

```sql
WITH latest AS (
  SELECT DISTINCT ON (allocator_id) allocator_id, engine_version, computed_at, mode, holding_flags
  FROM match_batches ORDER BY allocator_id, computed_at DESC
)
SELECT engine_version, mode,
  (jsonb_array_length(coalesce(holding_flags,'[]'::jsonb)) > 0) AS has_holding_flags,
  count(*) AS allocators,
  sum(jsonb_array_length(coalesce(holding_flags,'[]'::jsonb))) AS entries,
  min(computed_at) AS oldest_latest, max(computed_at) AS newest_latest
FROM latest GROUP BY 1,2,3 ORDER BY 1,2,3;
```

```sql
WITH latest AS (
  SELECT DISTINCT ON (allocator_id) id, allocator_id, engine_version, mode, holding_flags
  FROM match_batches ORDER BY allocator_id, computed_at DESC
)
SELECT l.engine_version, l.mode,
  (jsonb_array_length(coalesce(l.holding_flags,'[]'::jsonb)) > 0) AS has_holding_flags,
  count(*) AS candidates,
  count(*) FILTER (WHERE c.score_breakdown->'raw'->>'sharpe_lift' IS NULL) AS sharpe_lift_null,
  count(*) FILTER (WHERE c.score_breakdown->'raw'->>'dd_improvement' IS NULL) AS dd_improvement_null,
  count(*) FILTER (WHERE c.score_breakdown->'raw'->>'corr_with_portfolio' IS NULL) AS corr_with_portfolio_null,
  count(*) FILTER (WHERE c.score_breakdown->'portfolio_fit' IS NULL OR c.score_breakdown->>'portfolio_fit' IS NULL) AS portfolio_fit_null
FROM latest l JOIN match_candidates c ON c.batch_id = l.id AND c.exclusion_reason IS NULL
GROUP BY 1,2,3 ORDER BY 1,2,3;
```
