```ts
export type MemberProgressStatus = "waiting" | "in_process" | "successful" | "degraded";
export interface MemberProgressEntry {
  seq: number;
  exchange: string | null;
  label: string | null;
  status: MemberProgressStatus;
}
export type StitchJobStatus =
  | "pending" | "running" | "done" | "done_pending_children"
  | "failed_retry" | "failed_final";
export interface SyncProgressResponse {
  jobStatus: StitchJobStatus | null;   // null = no stitch_composite job visible for this strategy
  stalled: boolean;
  memberProgress: MemberProgressEntry[]; // [] until the worker's first write
}
export const STALL_THRESHOLD_MS = 600_000; // 10 min — see rationale in Task 2
```
