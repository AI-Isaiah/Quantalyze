// src/hooks/useStrategySyncPoller.ts
import type { ComputationStatus } from "@/components/strategy/SyncProgress"; // or the shared source in @/lib/types

export interface UseStrategySyncPollerOptions {
  enabled: boolean;                       // wizard: phase === 'waiting_for_complete'; SyncProgress: isActive
  strategyId: string;
  schedule: number | readonly number[];   // number → setInterval semantics (SyncProgress: 3000);
                                          // ladder → self-scheduled setTimeout walking then holding last (wizard: POLL_BACKOFF_MS)
  maxConsecutiveErrors?: number;          // wizard: 3; SyncProgress: undefined (NO escalation — pinned asymmetry)
  maxAttempts?: number;                   // SyncProgress: 40 (attempt N+1 escalates); wizard: undefined
  missingRowGracePolls?: number;          // SyncProgress: 10 (poll 11 escalates); wizard: undefined
  onStatus: (status: ComputationStatus, error: string | null) => void;
  onTerminal?: (status: ComputationStatus) => Promise<"done" | "repoll"> | "done" | "repoll";
                                          // wizard: heavy composite/single-key arms, R2-5 empty-series → "repoll";
                                          // SyncProgress: undefined (terminal is just a forwarded onStatus)
  onError: () => void;                    // escalation sink: wizard failPolling→SYNC_FAILED; SyncProgress onStatusChange("error")
}
export function useStrategySyncPoller(opts: UseStrategySyncPollerOptions): void;
