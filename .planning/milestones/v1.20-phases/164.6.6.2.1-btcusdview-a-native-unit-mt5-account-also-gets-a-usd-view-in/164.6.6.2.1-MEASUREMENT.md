"""D-09 spike: how does the REAL compose_allocator_equity treat a BTC key (already in USD)
whose USD series ends early / has an interior unpriced stretch?  Measurement only.
Usage: PYTHONPATH=. python btcusd_compose_probe.py m1|m2|m3
Every number is a literal printed below; hand values are computed independently of the
replay (closed forms), never by calling replay_key_equity/compose.
"""
from __future__ import annotations

import sys
from datetime import date, timedelta

import pandas as pd

from services.allocator_equity_compose import compose_allocator_equity


def iso(d0: str, n: int) -> list[str]:
    base = date.fromisoformat(d0)
    return [(base + timedelta(days=i)).isoformat() for i in range(n)]


def series(days, vals):
    return pd.Series([float(v) for v in vals], index=list(days))


def show(tag, payload):
    print(f"[{tag}] curve last 3 : {payload['curve'][-3:]}")
    print(f"[{tag}] returns last 3: {payload['returns'][-3:]}")
    print(f"[{tag}] flags         : {payload['flags']}")
    print(f"[{tag}] degrade       : {payload['degrade_reasons']}")
    print(f"[{tag}] is_trustworthy: {payload['is_trustworthy']}")
    print(f"[{tag}] scalars       : {payload['scalars']}")


# ---- shared literals --------------------------------------------------------------
A_R = 0.01          # key A (USD) constant daily return
B_R = 0.02          # key B (BTC key, already in USD) constant daily return
A_ANCHOR = 1000.0   # A: realized terminal == live anchor on D
B_TERM = 60000.0    # 1.0 BTC x 60000 (realized terminal USD on B's last USD return day)
B_ANCHOR = 61200.0  # 1.02 BTC x 60000 (D-17: live balance at the latest completed close)

D = "2026-06-10"
A_DAYS = iso("2026-06-06", 5)           # 06-06 .. 06-10 (ends D)


def a_level(day):  # closed form: A_ANCHOR / (1+A_R)^(days between day and D)
    k = (date.fromisoformat(D) - date.fromisoformat(day)).days
    return A_ANCHOR / (1.0 + A_R) ** k


def m1():
    print("== M1: BTC key B's USD series ends D-1 (live anchor priced at latest completed close)")
    b_days = iso("2026-06-05", 5)  # 06-05 .. 06-09 = D-1
    assert b_days[-1] == "2026-06-09" and A_DAYS[-1] == D
    p = compose_allocator_equity(
        {"A": series(A_DAYS, [A_R] * 5), "B": series(b_days, [B_R] * 5)},
        {"A": [], "B": []},
        {"A": A_ANCHOR, "B": B_ANCHOR},
        realized_terminal_by_key={"A": (D, A_ANCHOR), "B": ("2026-06-09", B_TERM)},
    )
    show("M1", p)
    curve = {r["date"]: r["equity_usd"] for r in p["curve"]}
    # hand: B is in D at its live anchor (carried level 61200) => curve(D) = A_D + 61200
    hand_D = a_level(D) + B_ANCHOR
    b_contrib_D = curve[D] - a_level(D)
    print(f"[M1] hand curve(D)={hand_D!r}  measured curve(D)={curve[D]!r}")
    print(f"[M1] B contribution on D (curve - A hand level) = {b_contrib_D!r}; B anchor = {B_ANCHOR!r}")
    print(f"[M1] B contribution on D-1 = {curve['2026-06-09'] - a_level('2026-06-09')!r}")
    ok = (
        abs(b_contrib_D - B_ANCHOR) < 1e-9
        and "missing_return_inside_coverage" not in p["flags"]
        and p["is_trustworthy"] is True  # no BLOCKING degrade reason (_BLOCKING_REASONS)
    )
    # A's book return on D vs the book's: B carried flat (r=0) on D dilutes the book return.
    book_r_D = {r["date"]: r["r"] for r in p["returns"]}[D]
    print(f"[M1] book return on D={book_r_D!r}; A alone={A_R!r}; hand (A_{{D-1}}*A_R)/(A_{{D-1}}+B_anchor)="
          f"{a_level('2026-06-09') * A_R / (a_level('2026-06-09') + B_ANCHOR)!r}")
    print(f"[M1] in_D_at_anchor={abs(b_contrib_D - B_ANCHOR) < 1e-9} "
          f"no_missing_token={'missing_return_inside_coverage' not in p['flags']} "
          f"trustworthy={p['is_trustworthy']} degrade={p['degrade_reasons']}")
    print("M1_RESULT", "PASS" if ok else "FAIL")


def m2():
    print("== M2: BTC key B's live anchor is unpriced -> trailing; B passed in departed_end_by_key={B: P}, P=D-2")
    p_day = "2026-06-08"  # P = D-2
    b_days = iso("2026-06-05", 4)  # 06-05 .. 06-08 = P
    assert b_days[-1] == p_day
    args = dict(
        flows_by_key={"A": [], "B": []},
        anchors_by_key={"A": A_ANCHOR, "B": B_TERM},  # B's anchor == its realized terminal level
    )
    with_b = compose_allocator_equity(
        {"A": series(A_DAYS, [A_R] * 5), "B": series(b_days, [B_R] * 4)},
        args["flows_by_key"], args["anchors_by_key"],
        departed_end_by_key={"B": p_day},
        realized_terminal_by_key={"A": (D, A_ANCHOR), "B": (p_day, B_TERM)},
    )
    show("M2 with B (T1 exit seam)", with_b)
    print("[M2] full curve:", with_b["curve"])
    print("[M2] full returns:", with_b["returns"])
    without_b = compose_allocator_equity(
        {"A": series(A_DAYS, [A_R] * 5)}, {"A": []}, {"A": A_ANCHOR},
        realized_terminal_by_key={"A": (D, A_ANCHOR)},
    )
    show("M2 without B (A alone)", without_b)
    # contrast: the same B WITHOUT departed_end_by_key (what the compose does by default)
    carried = compose_allocator_equity(
        {"A": series(A_DAYS, [A_R] * 5), "B": series(b_days, [B_R] * 4)},

        realized_terminal_by_key={"A": (D, A_ANCHOR), "B": (p_day, B_TERM)},

# ---- M3 literals -------------------------------------------------------------------
M3_DAYS = iso("2026-06-01", 10)  # 06-01 .. 06-10
M3_NATIVE = dict(zip(M3_DAYS, [1.00, 1.02, 1.01, 1.05, 1.04, 1.08, 1.07, 1.10, 1.12, 1.09]))  # BTC level
M3_CLOSE = dict(zip(M3_DAYS, [60000.0, 61500.0, 59000.0, 62000.0, None, 63000.0, 60500.0, 64000.0, 65000.0, 62500.0]))
H = "2026-06-05"   # h: the unpriced close
H1 = "2026-06-06"  # h+1: the converter also drops the day after an unpriced day


def m3():
    print("== M3: interior hole - B's USD returns absent on h=06-05 and h+1=06-06, NO new compose input")
    hand = {d: M3_NATIVE[d] * M3_CLOSE[d] for d in M3_DAYS if M3_CLOSE[d] is not None}  # native level x that day's close
    ret = {}
    for i in range(1, len(M3_DAYS)):
        d, prev = M3_DAYS[i], M3_DAYS[i - 1]
        if d in (H, H1):
            continue  # unpriced day / day after: no USD return
        ret[d] = hand[d] / hand[prev] - 1.0
    b_days = sorted(ret)
    print("[M3] hand USD levels (native x close):", {d: round(v, 6) for d, v in hand.items()})
    print("[M3] B return days:", b_days)
    terminal = hand[M3_DAYS[-1]]
    a_days = M3_DAYS[1:]  # A: 06-02 .. 06-10, r = 0.01, anchor 1000 on 06-10

    def a_lvl(day):
        return A_ANCHOR / (1.0 + A_R) ** (date.fromisoformat(M3_DAYS[-1]) - date.fromisoformat(day)).days

    p = compose_allocator_equity(
        {"A": series(a_days, [A_R] * len(a_days)), "B": series(b_days, [ret[d] for d in b_days])},
        {"A": [], "B": []},
        {"A": A_ANCHOR, "B": terminal},
        realized_terminal_by_key={"A": (M3_DAYS[-1], A_ANCHOR), "B": (M3_DAYS[-1], terminal)},
    )
    show("M3", p)
    curve = {r["date"]: r["equity_usd"] for r in p["curve"]}
    print("[M3] full curve:", {d: round(v, 6) for d, v in curve.items()})
    # B's contribution per curve day = curve - A hand level (independent of the replay)
    contrib = {d: curve[d] - a_lvl(d) for d in curve}
    print("[M3] B contribution by day (curve - A hand):", {d: round(v, 6) for d, v in contrib.items()})
    before_h = [d for d in M3_DAYS if d < H and d in contrib]
    diffs = {d: contrib[d] - hand[d] for d in before_h}
    print("[M3] levels BEFORE h: contribution - hand USD level:", {d: round(v, 9) for d, v in diffs.items()})
    prev_replayed = max(d for d in b_days if d < H)
    carried_on_h = abs(contrib[H] - contrib[prev_replayed]) < 1e-9
    print(f"[M3] B contribution on h={contrib[H]!r}; on {prev_replayed}={contrib[prev_replayed]!r}; "
          f"carried_on_h(r=0 at the last priced replay level)={carried_on_h}; "
          f"hand USD level on h = unpriced (no close); hand on h-1={hand[prev_replayed]!r}")
    print(f"[M3] missing_return_inside_coverage token present: {'missing_return_inside_coverage' in p['flags']}")
    worst = max(abs(v) for v in diffs.values())
    stale = carried_on_h or worst > 1e-9
    print(f"[M3] worst |level - hand| before h = {worst!r}; carried_on_h={carried_on_h}")
    print("M3_RESULT", "STALE-CARRY" if stale else "CLEAN")


def m4():
    """The book's latest day: the MT5 key-mode branch of derive_broker_dailies builds its per-key returns
    with combine_mt5_deal_ledger (which gap-fills only INSIDE [first deal day, last deal day]) and its
    realized terminal day from mt5_day_pnl(...).index[-1] (the last deal/flow day).  The allocator compose
    reads csv_daily_returns as stored.  Nothing extends a key to the run's UTC day.  Run the real functions."""
    from datetime import datetime, timezone

    from services.broker_dailies import combine_mt5_deal_ledger, mt5_day_pnl

    def epoch(day: str) -> int:
        d = date.fromisoformat(day)
        return int(datetime(d.year, d.month, d.day, 12, tzinfo=timezone.utc).timestamp())

    def deals_for(trading_days, deposit_day):
        deals = [{"type": 2, "profit": 1000.0, "swap": 0.0, "commission": 0.0, "fee": 0.0, "time": epoch(deposit_day)}]
        for dd in trading_days:
            deals.append({"type": 1, "entry": 1, "profit": 10.0, "swap": 0.0, "commission": 0.0, "fee": 0.0, "time": epoch(dd)})
        return deals

    run_day = "2026-06-10"  # the run's UTC day
    for tag, trading_days in (
        ("deal TODAY (a trading deal dated the run's UTC day)", ["2026-06-07", "2026-06-08", "2026-06-10"]),
        ("NO deal today (last trading deal two days before the run's UTC day)", ["2026-06-05", "2026-06-06", "2026-06-08"]),
    ):
        deals = deals_for(trading_days, "2026-06-04")
        eq = 1000.0 + 10.0 * len(trading_days)
        returns, _ = combine_mt5_deal_ledger(deals, account_equity=eq, account_balance=eq)
        pnl = mt5_day_pnl(deals)
        rr = {d.date().isoformat(): float(v) for d, v in returns.dropna().items()}  # the writer stores only defined returns
        iso_ret = pd.Series(list(rr.values()), index=list(rr.keys()))
        pay = compose_allocator_equity({"K": iso_ret}, {"K": []}, {"K": eq})  # flows omitted: only the last date is measured
        last = pay["curve"][-1]["date"]
        print(f"[M4] {tag}")
        print(f"[M4]   raw returns (NaN = day the TWR leaves out): {[(d.date().isoformat(), None if v != v else round(float(v), 6)) for d, v in returns.items()]}")
        print(f"[M4]   run UTC day={run_day}; returns index last={returns.index[-1].date().isoformat()}; "
              f"mt5_day_pnl index last={pnl.index[-1].date().isoformat()}; composite curve last date={last}")
        print(f"[M4]   curve last date == run day: {last == run_day}; == last deal day: "
              f"{last == max(trading_days)}")
    print("M4_NOTE: the last curve date follows the last stored return day (= the last deal day); it equals "
          "the run's UTC day only when a deal is dated that day.")
