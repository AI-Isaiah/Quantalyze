```sql
SELECT shobj_description(oid, 'pg_database') AS which_database
  FROM pg_database WHERE datname = current_database();

SELECT
  count(*) FILTER (WHERE EXISTS (
    SELECT 1 FROM public.compute_jobs p
     WHERE p.id = ANY (c.parent_job_ids) AND p.status = 'failed_final'
  )) AS children_with_a_failed_final_parent,
  count(*) FILTER (WHERE NOT EXISTS (
    SELECT 1 FROM unnest(c.parent_job_ids) AS pid
     WHERE NOT EXISTS (
       SELECT 1 FROM public.compute_jobs p WHERE p.id = pid AND p.status = 'done'
     )
  )) AS children_whose_parents_are_all_done
FROM public.compute_jobs c
WHERE c.status = 'done_pending_children';
```
