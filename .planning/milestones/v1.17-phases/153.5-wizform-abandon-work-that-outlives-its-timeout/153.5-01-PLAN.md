_MT5_TERMINAL_EPOCHS: dict[str, int] = {}   # deliberately NOT Final — it is mutated

def _mt5_epoch_for(terminal_key: str) -> int:
    # dict.get with a default, NEVER setdefault — a read must not mint an entry
    return _MT5_TERMINAL_EPOCHS.get(terminal_key, 0)

def bump_mt5_terminal_epoch(terminal_key: str) -> int:
