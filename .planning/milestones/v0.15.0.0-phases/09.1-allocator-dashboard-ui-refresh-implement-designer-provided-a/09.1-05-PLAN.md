**WidgetGrid.tsx** — the host component. Signature:

```tsx
"use client";

import { useRef, useState } from "react";
import type { TileConfig } from "../lib/types";
import { WidgetChrome } from "./WidgetChrome";

type Props = {
  tiles: TileConfig[];
  onResize: (k: string, w: 1 | 2 | 3 | 4) => void;
  onRemove: (k: string) => void;
  onMove: (fromK: string, toK: string) => void;
  renderWidget: (k: string) => React.ReactNode;
};

export function WidgetGrid({ tiles, onResize, onRemove, onMove, renderWidget }: Props) {
  const [draggingK, setDraggingK] = useState<string | null>(null);
  const [dragOverK, setDragOverK] = useState<string | null>(null);
  const gridRef = useRef<HTMLDivElement>(null);

  return (
    <div
      ref={gridRef}
      className="widget-grid"
      style={{
        display: "grid",
        gridTemplateColumns: "repeat(4, minmax(0, 1fr))",
        gap: 10,
        marginTop: 10,
      }}
    >
      {tiles.map(({ k, w }) => (
        <WidgetCell
          key={k}
          k={k}
          w={w}
          draggingK={draggingK}
          dragOverK={dragOverK}
          setDraggingK={setDraggingK}
          setDragOverK={setDragOverK}
          onResize={onResize}
          onRemove={onRemove}
          onMove={onMove}
          renderWidget={renderWidget}
          gridRef={gridRef}
        />
      ))}
      <style jsx>{`
        .widget-grid :global(.widget-cell:hover) .widget-chrome,
        .widget-grid :global(.widget-cell:focus-within) .widget-chrome {
          opacity: 1 !important;
        }
        @media (max-width: 980px) { .widget-grid { grid-template-columns: repeat(2, minmax(0, 1fr)) !important; } }
        @media (max-width: 640px) { .widget-grid { grid-template-columns: 1fr !important; } }
      `}</style>
    </div>
  );
}

function WidgetCell(props: {
  k: string; w: 1 | 2 | 3 | 4;
  draggingK: string | null;
  dragOverK: string | null;
  setDraggingK: (v: string | null) => void;
  setDragOverK: (v: string | null) => void;
  onResize: (k: string, w: 1 | 2 | 3 | 4) => void;
  onRemove: (k: string) => void;
  onMove: (fromK: string, toK: string) => void;
  renderWidget: (k: string) => React.ReactNode;
  gridRef: React.RefObject<HTMLDivElement>;
}) {
  const { k, w, draggingK, dragOverK, setDraggingK, setDragOverK, onResize, onRemove, onMove, renderWidget, gridRef } = props;

  // D-01 pointer-resize — port from designer-bundle/project/src/widget-grid.jsx:100-135
  // Structure: absolute-positioned .resize-handle div (6px wide, full height, cursor: col-resize).
  // onPointerDown captures the pointer, onPointerMove computes target span from delta relative to
  // column width, snaps to nearest [1,2,3,4] on onPointerUp, calls onResize(k, nextW).
  function handleResizePointerDown(e: React.PointerEvent<HTMLDivElement>) {
    e.stopPropagation();
    e.preventDefault();
    const startX = e.clientX;
    const startW = w;
    const gridEl = gridRef.current;
    const gridWidth = gridEl?.getBoundingClientRect().width ?? 0;
    const gap = 10;
    // Column width in pixels: (gridWidth - 3*gap) / 4
    const colWidth = gridWidth > 0 ? (gridWidth - 3 * gap) / 4 : 240;
    (e.currentTarget as HTMLDivElement).setPointerCapture(e.pointerId);

    function move(evt: PointerEvent) {
      const delta = evt.clientX - startX;
      const deltaCols = delta / (colWidth + gap);
      const target = Math.max(1, Math.min(4, Math.round(startW + deltaCols))) as 1 | 2 | 3 | 4;
      // Visual preview: write target via inline style? For MVP, defer visual feedback to final snap on pointerup.
      (e.currentTarget as HTMLDivElement).dataset.pendingW = String(target);
    }
    function up(evt: PointerEvent) {
      const target = Number((e.currentTarget as HTMLDivElement).dataset.pendingW ?? startW) as 1 | 2 | 3 | 4;
      if (target !== startW) onResize(k, target);
      window.removeEventListener("pointermove", move);
      window.removeEventListener("pointerup", up);
    }
    window.addEventListener("pointermove", move);
    window.addEventListener("pointerup", up);
  }

  return (
    <div
      data-widget-id={k}
      draggable
      onDragStart={() => setDraggingK(k)}
      onDragEnd={() => { setDraggingK(null); setDragOverK(null); }}
      onDragOver={(e) => { e.preventDefault(); setDragOverK(k); }}
      onDrop={(e) => {
        e.preventDefault();
        if (draggingK && draggingK !== k) onMove(draggingK, k);
        setDraggingK(null); setDragOverK(null);
      }}
      className="widget-cell"
      style={{
        gridColumn: `span ${w}`,
        minHeight: 140,
        position: "relative",
        transition: "box-shadow 120ms ease",
        boxShadow: dragOverK === k && draggingK !== k ? "0 0 8px rgba(27,107,90,0.4)" : undefined,
        opacity: draggingK === k ? 0.55 : 1,
      }}
    >
      <WidgetChrome
        k={k}
        w={w}
        onResize={onResize}
        onRemove={onRemove}
        onMove={onMove}
      />
      {/* D-01 pointer-resize handle (right edge, full height) */}
      <div
        className="resize-handle"
        onPointerDown={handleResizePointerDown}
        aria-hidden="true"
        style={{
          position: "absolute",
          top: 0,
          right: 0,
          width: 6,
          height: "100%",
          cursor: "col-resize",
          zIndex: 1,
        }}
      />
      {renderWidget(k)}
    </div>
  );
}
```

**WidgetChrome.tsx** — hover/focus-within chrome wrapper with SizeStepper + drag handle (keyboard-reorder capable) + close button + always-present overflow menu:

```tsx
"use client";

import { useEffect, useRef, useState } from "react";
import { SizeStepper } from "./SizeStepper";

type Props = {
  k: string;
  w: 1 | 2 | 3 | 4;
  onResize: (k: string, w: 1 | 2 | 3 | 4) => void;
  onRemove: (k: string) => void;
  onMove: (fromK: string, toK: string) => void;  // V1: used by keyboard-reorder + overflow menu; callers compute neighbor
};

export function WidgetChrome({ k, w, onResize, onRemove, onMove }: Props) {
  // D-04 (V1 accepted): keyboard-reorder mode.
  // Enter/Space toggles keyboard-reorder mode on the drag handle;
  // ArrowUp/ArrowDown invoke onMove(k, "prev" | "next"); Esc exits the mode.
  const [kbdMode, setKbdMode] = useState(false);
  const [menuOpen, setMenuOpen] = useState(false);
  const menuRef = useRef<HTMLDivElement>(null);

  // We signal move intent via a sentinel "prev" / "next" string; the parent
  // (AllocationDashboardV2) maps this to moveWidget(fromK, toK) by computing
  // the adjacent tile's `k` from the current config.tiles order.
  function handleDragKeyDown(e: React.KeyboardEvent<HTMLButtonElement>) {
    if (e.key === "Enter" || e.key === " ") {
      e.preventDefault();
      setKbdMode(v => !v);
      return;
    }
    if (!kbdMode) return;
    if (e.key === "ArrowUp") {
      e.preventDefault();
      onMove(k, "prev" as unknown as string);   // parent resolves "prev" → actual neighbor k
    } else if (e.key === "ArrowDown") {
      e.preventDefault();
      onMove(k, "next" as unknown as string);   // parent resolves "next" → actual neighbor k
    } else if (e.key === "Escape") {
      e.preventDefault();
      setKbdMode(false);
    }
  }

  // Outside-click dismiss for the overflow menu
  useEffect(() => {
    if (!menuOpen) return;
    const handler = (e: MouseEvent) => {
      if (menuRef.current?.contains(e.target as Node)) return;
      setMenuOpen(false);
    };
    document.addEventListener("click", handler);
    return () => document.removeEventListener("click", handler);
  }, [menuOpen]);

  return (
    <div
      className="widget-chrome"
      style={{
        position: "absolute",
        top: 8, right: 8,
        display: "flex",
        gap: 4,
        opacity: 0,
        transition: "opacity 120ms ease",
        zIndex: 2,
      }}
    >
      <SizeStepper current={w} onChange={(next) => onResize(k, next)} />
      <button
        type="button"
        role="button"
        aria-label="Reorder widget"
        aria-pressed={kbdMode}
        onKeyDown={handleDragKeyDown}
        className="chrome-btn drag-handle"
        style={{
          cursor: "grab",
          padding: "2px 6px",
          fontSize: 11,
          background: kbdMode ? "var(--accent)" : "transparent",
          color: kbdMode ? "white" : "var(--text-secondary)",
          border: "1px solid var(--border)",
          borderRadius: 3,
        }}
      >
        ⠇
      </button>
      <button
        type="button"
        aria-label={`Remove ${k} widget`}
        onClick={() => onRemove(k)}
        className="chrome-btn close-btn"
        style={{ padding: "2px 6px", fontSize: 11 }}
      >
        ×
      </button>

      {/* Always-present overflow menu — screen-reader path for reorder + remove.
          Not hover-only; visible alongside the rest of the chrome when chrome is revealed. */}
      <div ref={menuRef} style={{ position: "relative" }}>
        <button
          type="button"
          aria-label="Widget options"
          aria-haspopup="menu"
          aria-expanded={menuOpen}
          onClick={() => setMenuOpen(v => !v)}
          className="chrome-btn overflow-btn"
          style={{ padding: "2px 6px", fontSize: 11 }}
        >
          ⋯
        </button>
        {menuOpen && (
          <div
            role="menu"
            aria-label="Widget actions"
            style={{
              position: "absolute",
              top: "calc(100% + 4px)",
              right: 0,
              background: "var(--surface)",
              border: "1px solid var(--border)",
              borderRadius: 4,
              padding: 4,
              minWidth: 120,
              zIndex: 5,
            }}
          >
            <button
              type="button"
              role="menuitem"
              onClick={() => { onMove(k, "prev" as unknown as string); setMenuOpen(false); }}
              style={{ display: "block", width: "100%", textAlign: "left", padding: "4px 8px", fontSize: 12, background: "transparent", border: "none", cursor: "pointer" }}
            >
              Move up
            </button>
            <button
              type="button"
              role="menuitem"
              onClick={() => { onMove(k, "next" as unknown as string); setMenuOpen(false); }}
              style={{ display: "block", width: "100%", textAlign: "left", padding: "4px 8px", fontSize: 12, background: "transparent", border: "none", cursor: "pointer" }}
            >
              Move down
            </button>
            <button
              type="button"
              role="menuitem"
              onClick={() => { onRemove(k); setMenuOpen(false); }}
              style={{ display: "block", width: "100%", textAlign: "left", padding: "4px 8px", fontSize: 12, color: "var(--danger, #DC2626)", background: "transparent", border: "none", cursor: "pointer" }}
            >
              Remove
            </button>
          </div>
        )}
      </div>
    </div>
  );
}
```

```tsx
import { useEffect, useRef, useState } from "react";

  onPick: (k: string) => void;

export function WidgetPicker({ isOpen, onClose, anchorRef, activeKeys, onPick }: Props) {
  const [query, setQuery] = useState("");
  const popoverRef = useRef<HTMLDivElement>(null);

  // Outside-click dismiss (matches designer widget-grid.jsx:248-259)
  useEffect(() => {
    if (!isOpen) return;
    const handler = (e: MouseEvent) => {
      if (popoverRef.current?.contains(e.target as Node)) return;
      if (anchorRef.current?.contains(e.target as Node)) return;
      onClose();
    };
    document.addEventListener("click", handler);
    return () => document.removeEventListener("click", handler);
  }, [isOpen, onClose, anchorRef]);

  // Esc dismiss (clone focus-trap idiom from AddWidgetModal.tsx:29-62)

    if (!isOpen) return;
    const onKey = (e: KeyboardEvent) => { if (e.key === "Escape") onClose(); };
    document.addEventListener("keydown", onKey);
    return () => document.removeEventListener("keydown", onKey);
  }, [isOpen, onClose]);

  if (!isOpen) return null;

  const entries = Object.values(WIDGET_REGISTRY);
  const filtered = query
    ? entries.filter(e => e.name.toLowerCase().includes(query.toLowerCase()) || e.id.toLowerCase().includes(query.toLowerCase()))
    : entries;
  const byCategory = filtered.reduce<Record<string, typeof entries>>((acc, e) => {
    (acc[e.category] ||= []).push(e);
    return acc;
  }, {});

  return (

        placeholder="Search widgets…"
        value={query}
        onChange={(e) => setQuery(e.target.value)}
        aria-label="Search widgets"
        autoFocus
        style={{ width: "100%", padding: "6px 8px", marginBottom: 8, border: "1px solid var(--border)", borderRadius: 4 }}
      />
      {Object.entries(byCategory).map(([cat, items]) => (
        <section key={cat} aria-label={cat}>
          <h4 style={{ fontSize: 11, textTransform: "uppercase", letterSpacing: "0.08em", color: "var(--text-muted)", margin: "12px 0 4px" }}>

                key={w.id}
                type="button"
                disabled={active}
                onClick={() => { onPick(w.id); onClose(); }}
```
