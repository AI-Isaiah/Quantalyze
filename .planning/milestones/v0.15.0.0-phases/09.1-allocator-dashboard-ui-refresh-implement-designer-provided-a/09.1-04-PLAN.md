```ts
import { buildHoldingRef } from "./holding-outcome-adapter";
// Import FlaggedHolding, StrategyRow types from queries (exact names — grep for them).

export interface HoldingsAdapterInputs {
  holdingsSummary: Array<{
    venue: string;
    symbol: string;
    holding_type: "spot" | "derivative";
    quantity: number;
    value_usd: number;
    api_key_id?: string;
    allocated_at?: string | null;          // ISO date (if present in payload; null-safe)
  }>;
  flaggedHoldings: Array<{
    venue: string;
    symbol: string;
    holding_type: "spot" | "derivative";
    composite_score: number;
    top_candidate_strategy_id?: string | null;
  }>;
  matchDecisionsByHoldingRef: Record<string, { id: string } | null>;
  strategies: Array<{
    id: string;
    name: string;
    alias?: string | null;
    codename?: string | null;
    strategy_types?: string[] | null;
    strategy_analytics?: {
      sharpe?: number | null;
      max_drawdown?: number | null;
      mtd?: number | null;
    } | null;
  }>;
  /**
   * Optional pre-keyed analytics (if caller has already denormalized).
   * If provided, takes precedence over strategies[].strategy_analytics.
   */
  analyticsByStrategyId?: Record<string, {
    sharpe: number | null; max_drawdown: number | null; mtd: number | null;
  }>;
  /**
   * D-18 (R1 accepted): caller-supplied holding→strategy correspondence.
   * Keyed by `buildHoldingRef(venue, symbol, holding_type)`. When absent OR
   * when a row's ref has no entry, `strategy` resolves to null. The adapter
   * does NOT invent joins (no asset_symbols heuristic); HoldingsTabPanel
   * (Plan 08) builds this map by porting the legacy `enrichedHoldings`
   * correspondence from AllocationDashboard.tsx:697-724.
   */
  holdingToStrategyId?: Record<string, string>;
  /** Injectable "now" for age computation; defaults to new Date(). */
  now?: Date;
}

export interface DesignHoldingRow {
  id: string;
  venue: string;
  symbol: string;
  holding_type: "spot" | "derivative";
  strategy: string | null;
  manager: string | null;
  tag: string | null;
  alloc: number;
  weight: number;
  mtd: number | null;
  sharpe: number | null;
  dd: number | null;
  age: number | null;
  status: "underperform" | "watch" | "ok";
  bridgeCandidate: boolean;
}

const FLAG_COMPOSITE_THRESHOLD = 50;  // Phase 09 D-05 (see 09-02 SUMMARY)
const WATCH_COMPOSITE_THRESHOLD = 40; // watch = near-threshold

export function toDesignHoldings(inputs: HoldingsAdapterInputs): DesignHoldingRow[] {
  const { holdingsSummary, flaggedHoldings, matchDecisionsByHoldingRef, strategies, analyticsByStrategyId, holdingToStrategyId, now } = inputs;

  // Σ value_usd for weight denominator
  const total = holdingsSummary.reduce((sum, h) => sum + (Number.isFinite(h.value_usd) ? h.value_usd : 0), 0);

  // Index flaggedHoldings by ref for O(1) lookup
  const flaggedByRef = new Map<string, typeof flaggedHoldings[number]>();
  for (const f of flaggedHoldings) flaggedByRef.set(buildHoldingRef(f), f);

  // Index strategies by id for O(1) lookup.
  const strategiesById = new Map<string, typeof strategies[number]>();
  for (const s of strategies) strategiesById.set(s.id, s);

  /**
   * Strategy join: caller-supplied map ONLY (D-18, R1 accepted).
   * No heuristic matching. No asset_symbols lookup.
   */
  function findStrategy(ref: string): typeof strategies[number] | null {
    const sid = holdingToStrategyId?.[ref];
    if (!sid) return null;
    return strategiesById.get(sid) ?? null;
  }

  const nowMs = (now ?? new Date()).getTime();

  return holdingsSummary.map(h => {
    const ref = buildHoldingRef(h);
    const flag = flaggedByRef.get(ref);
    const strat = findStrategy(ref);
    const analytics = strat
      ? (analyticsByStrategyId?.[strat.id] ?? {
          sharpe: strat.strategy_analytics?.sharpe ?? null,
          max_drawdown: strat.strategy_analytics?.max_drawdown ?? null,
          mtd: strat.strategy_analytics?.mtd ?? null,
        })
      : { sharpe: null, max_drawdown: null, mtd: null };

    const weight = total > 0 ? h.value_usd / total : 0;

    let status: DesignHoldingRow["status"] = "ok";
    if (flag) {
      status = flag.composite_score >= FLAG_COMPOSITE_THRESHOLD
        ? "underperform"
        : (flag.composite_score >= WATCH_COMPOSITE_THRESHOLD ? "watch" : "ok");
    }

    const ageMs = h.allocated_at ? nowMs - new Date(h.allocated_at).getTime() : null;
    const age = ageMs != null && ageMs > 0 ? Math.floor(ageMs / 86_400_000) : null;

    const bridgeCandidate = matchDecisionsByHoldingRef[ref] != null;

    return {
      id: ref,
      venue: h.venue,
      symbol: h.symbol,
      holding_type: h.holding_type,
      strategy: strat ? (strat.alias ?? strat.name) : null,
      manager: strat?.codename ?? null,
      tag: strat?.strategy_types?.[0] ?? null,
      alloc: h.value_usd,
      weight,
      mtd: analytics.mtd,
      sharpe: analytics.sharpe,
      dd: analytics.max_drawdown,
      age,
      status,
      bridgeCandidate,
    };
  });
}
```
