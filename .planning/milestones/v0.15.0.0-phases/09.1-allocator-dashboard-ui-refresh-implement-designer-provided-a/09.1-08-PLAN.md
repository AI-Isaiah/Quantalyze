**OutcomeForm.tsx** — unified allocated/rejected form with a 3-option segmented control. The third option ("Modified") is a disabled placeholder per C1 accepted — visible in the UI so users see the future capability, disabled because `bridge-outcome-schema` does not yet accept that kind:

```tsx
"use client";

import { useState } from "react";
import type { DesignHoldingRow } from "../lib/holdings-adapter";
import { ALLOCATED_FIELDS, REJECTED_FIELDS, postBridgeOutcome } from "@/lib/bridge-outcome-schema";

type Action = "allocated" | "rejected";  // "modified" is rendered as a disabled mode, not a submittable state

/**
 * C1 accepted: expose a 3-option segmented control. "Modified (coming soon)"
 * is disabled + aria-disabled + carries an explanatory tooltip. This tells
 * allocators the capability is intended and pending a schema extension —
 * vs. silently dropping it from the UI (which hid the intent).
 */
const MODES: Array<{ key: Action | "modified"; label: string; disabled: boolean; tooltip?: string }> = [
  { key: "allocated", label: "Allocated", disabled: false },
  { key: "rejected",  label: "Rejected",  disabled: false },
  { key: "modified",  label: "Modified (coming soon)", disabled: true, tooltip: "Schema extension pending — see follow-up phase" },
];

type Props = {
  strategyId: string;
  row: DesignHoldingRow;
  onRecorded?: (id: string) => void;
};

export function OutcomeForm({ strategyId, row, onRecorded }: Props) {
  const [action, setAction] = useState<Action>("allocated");
  const [size, setSize] = useState<string>("");
  const [effectiveAt, setEffectiveAt] = useState<string>(new Date().toISOString().slice(0, 10));
  const [note, setNote] = useState<string>("");
  const [error, setError] = useState<string | null>(null);
  const [submitting, setSubmitting] = useState<boolean>(false);
  const [recordedId, setRecordedId] = useState<string | null>(null);

  async function handleSubmit(e: React.FormEvent) {
    e.preventDefault();
    setError(null);
    const schema = action === "allocated" ? ALLOCATED_FIELDS : REJECTED_FIELDS;
    const pctNum = Number(size);
    const values = action === "allocated"
      ? { percent_allocated: pctNum, allocated_at: effectiveAt, note: note || null }
      : { rejected_at: effectiveAt, note: note || null, reason: note || "" };
    const parsed = schema.safeParse(values);
    if (!parsed.success) { setError(parsed.error.issues[0]?.message ?? "Invalid input"); return; }
    setSubmitting(true);
    const result = await postBridgeOutcome({ strategyId, kind: action, values: parsed.data });
    setSubmitting(false);
    if (!result.ok) { setError(result.error); return; }
    setRecordedId(result.outcome.id);
    onRecorded?.(result.outcome.id);
  }

  if (recordedId) {
    return (
      <div role="status" className="rounded-md border border-green-300 bg-green-50 p-4">
        <div className="text-sm font-medium text-green-800">Outcome recorded</div>
        <div className="text-xs text-green-700 mt-1">Logged against {row.strategy ?? row.symbol}.</div>
      </div>
    );
  }

  return (
    <form onSubmit={handleSubmit} className="grid gap-3">
      <div role="group" aria-label="Action" className="flex gap-1">
        {MODES.map(mode => (
          <button
            key={mode.key}
            type="button"
            aria-pressed={!mode.disabled && action === mode.key}
            aria-disabled={mode.disabled}
            disabled={mode.disabled}
            title={mode.tooltip}
            onClick={() => { if (!mode.disabled) setAction(mode.key as Action); }}
            className={
              mode.disabled
                ? "px-3 py-1 text-xs rounded border border-border text-text-muted cursor-not-allowed opacity-60"
                : (action === mode.key
                    ? "px-3 py-1 text-xs bg-accent text-white rounded"
                    : "px-3 py-1 text-xs rounded border border-border")
            }
          >
            {mode.label}
          </button>
        ))}
      </div>
      {action === "allocated" && (
        <label className="text-xs grid gap-1">
          Allocation size (%)
          <input type="number" min="0" max="100" step="0.1" value={size} onChange={e => setSize(e.target.value)} required />
        </label>
      )}
      <label className="text-xs grid gap-1">
        Effective date
        <input type="date" value={effectiveAt} onChange={e => setEffectiveAt(e.target.value)} required />
      </label>
      <label className="text-xs grid gap-1">
        Note {action === "rejected" ? "(reason)" : "(optional)"}
        <textarea rows={2} value={note} onChange={e => setNote(e.target.value)} />
      </label>
      {error && <div role="alert" className="text-xs text-red-600">{error}</div>}
      <button type="submit" disabled={submitting} className="self-start px-3 py-1.5 text-sm bg-accent text-white rounded">
        {submitting ? "Submitting…" : "Record outcome"}
      </button>
    </form>
  );
}
```
