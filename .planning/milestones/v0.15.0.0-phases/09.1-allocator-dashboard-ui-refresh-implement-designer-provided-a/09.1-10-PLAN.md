```tsx
"use client";
import Link from "next/link";
import type { MyAllocationDashboardPayload } from "@/lib/queries";

type MandateSnapshot = {
  maxWeight?: number | null;
  correlationCeiling?: number | null;
  styleExclusions?: string[] | null;
  // whatever fields are on the payload
};

function MandateSnapshotCard({ mandate }: { mandate: MandateSnapshot | null }) {
  if (!mandate) {
    return (
      <div className="rounded-lg border border-border bg-surface p-4">
        <div className="text-xs uppercase text-text-muted">Mandate</div>
        <div className="mt-1 text-sm text-text-secondary">No mandate set yet.</div>
      </div>
    );
  }
  return (
    <div className="rounded-lg border border-border bg-surface p-4">
      <div className="text-xs uppercase text-text-muted">Mandate snapshot</div>
      <dl className="mt-2 grid grid-cols-1 sm:grid-cols-2 gap-2 text-sm">
        {mandate.maxWeight != null && <div><dt className="text-text-muted text-xs">Max weight</dt><dd>{(mandate.maxWeight * 100).toFixed(1)}%</dd></div>}
        {mandate.correlationCeiling != null && <div><dt className="text-text-muted text-xs">Correlation ceiling</dt><dd>{mandate.correlationCeiling.toFixed(2)}</dd></div>}
        {mandate.styleExclusions?.length ? (
          <div className="col-span-2"><dt className="text-text-muted text-xs">Style exclusions</dt><dd>{mandate.styleExclusions.join(", ")}</dd></div>
        ) : null}
      </dl>
    </div>
  );
}

export function MandateTabPanel(props: MyAllocationDashboardPayload) {
  const mandate = (props as any).mandate ?? null;  // field name may differ — match actual payload shape
  return (
    <div data-tab-panel="mandate" className="grid gap-4">
      <MandateSnapshotCard mandate={mandate} />
      <div className="rounded-lg border border-border bg-surface p-4 text-sm">
        <p className="mb-2">Edit your mandate in the profile surface. Changes take effect immediately across the Bridge engine.</p>
        <Link href="/profile?tab=mandate" className="text-accent hover:underline">
          Open Mandate form →
        </Link>
      </div>
    </div>
  );
}
```

**RiskTabPanel.tsx** — curated 6-widget grid:

```tsx
"use client";
import type { MyAllocationDashboardPayload } from "@/lib/queries";
import { WIDGET_COMPONENTS } from "./widgets";

const RISK_WIDGETS = [
  "var-expected-shortfall",
  "tail-risk",
  "risk-decomposition",
  "alpha-beta-decomposition",
  "regime-detector",
  "correlation-matrix",
] as const;

export function RiskTabPanel(props: MyAllocationDashboardPayload) {
  return (
    <div data-tab-panel="risk" className="grid grid-cols-1 lg:grid-cols-2 gap-4">
      {RISK_WIDGETS.map((id) => {
        const Component = WIDGET_COMPONENTS[id];
        if (!Component) {
          return (
            <div key={id} className="rounded-lg border border-border bg-surface p-4 text-xs text-text-muted">
              Widget unavailable: {id}
            </div>
          );
        }
        return (
          <div key={id} data-widget-id={id} className="rounded-lg border border-border bg-surface p-4">
            <Component data={props as any} timeframe="YTD" width={0} height={0} />
          </div>
        );
      })}
    </div>
  );
}
```
