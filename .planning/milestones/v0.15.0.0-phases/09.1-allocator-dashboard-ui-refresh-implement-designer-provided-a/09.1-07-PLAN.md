**CustomRangePicker.tsx:** port the designer's range-picker.jsx into a popover component.

```tsx
"use client";

import { useEffect, useRef, useState } from "react";

type Props = {
  isOpen: boolean;
  onClose: () => void;
  onApply: (range: { start: string; end: string }) => void;
  min: Date;
  max: Date;
  initialRange?: { start: string; end: string } | null;
};

export function CustomRangePicker({ isOpen, onClose, onApply, min, max, initialRange }: Props) {
  const [start, setStart] = useState<string>(initialRange?.start ?? min.toISOString().slice(0, 10));
  const [end, setEnd] = useState<string>(initialRange?.end ?? max.toISOString().slice(0, 10));
  const ref = useRef<HTMLDivElement>(null);

  useEffect(() => {
    if (!isOpen) return;
    const onKey = (e: KeyboardEvent) => { if (e.key === "Escape") onClose(); };
    const onClick = (e: MouseEvent) => {
      if (ref.current && !ref.current.contains(e.target as Node)) onClose();
    };
    document.addEventListener("keydown", onKey);
    document.addEventListener("click", onClick);
    return () => {
      document.removeEventListener("keydown", onKey);
      document.removeEventListener("click", onClick);
    };
  }, [isOpen, onClose]);

  if (!isOpen) return null;

  function apply() {
    if (start > end) return;
    onApply({ start, end });
    onClose();
  }

  const minIso = min.toISOString().slice(0, 10);
  const maxIso = max.toISOString().slice(0, 10);

  return (
    <div
      ref={ref}
      role="dialog"
      aria-label="Custom date range"
      style={{
        position: "absolute", top: "calc(100% + 6px)", right: 0, zIndex: 50,
        background: "var(--surface)", border: "1px solid var(--border)", borderRadius: 8,
        padding: 12, boxShadow: "0 4px 16px rgba(0,0,0,0.08)", minWidth: 280,
      }}
    >
      <div style={{ display: "flex", gap: 8, flexDirection: "column" }}>
        <label style={{ fontSize: 11, color: "var(--text-muted)" }}>
          Start
          <input type="date" value={start} min={minIso} max={maxIso} onChange={(e) => setStart(e.target.value)} />
        </label>
        <label style={{ fontSize: 11, color: "var(--text-muted)" }}>
          End
          <input type="date" value={end} min={minIso} max={maxIso} onChange={(e) => setEnd(e.target.value)} />
        </label>
      </div>
      <div style={{ display: "flex", gap: 6, marginTop: 12, justifyContent: "flex-end" }}>
        <button type="button" onClick={onClose} style={{ fontSize: 12, padding: "4px 10px" }}>Cancel</button>
        <button
          type="button"
          onClick={apply}
          disabled={start > end}
          style={{
            fontSize: 12, padding: "4px 10px",
            background: start > end ? "transparent" : "var(--accent)",
            color: start > end ? "var(--text-muted)" : "white",
            border: "1px solid var(--accent)", borderRadius: 4,
          }}
        >
          Apply
        </button>
      </div>
    </div>
  );
}
```
