STABLECOINS = {"USDT", "USDC", "BUSD", "DAI", "TUSD", "USD"}
RAW_PAYLOAD_CAP_BYTES = 4096  # D-02 / Claude's discretion ~4KB


class DeribitNotSupportedError(ccxt.NotSupported):
    """f3 Path B: raised when the allocator worker is asked to fetch spot from Deribit.

    Subclasses ccxt.NotSupported so any catch-all `except ccxt.NotSupported` still
    works. `_map_exception_to_sync_status` maps it to 'error' (not 'revoked') so the

def _cap_raw_payload(payload: dict) -> dict:
    """Truncate raw_payload JSON to ~4KB; if over, replace with {'truncated': True, 'preview': str[:3900]}."""
    encoded = json.dumps(payload, default=str)
    if len(encoded) <= RAW_PAYLOAD_CAP_BYTES:
        return payload
    return {"truncated": True, "preview": encoded[:3900]}


def _map_exception_to_sync_status(exc: Exception) -> str:
    """
    INGEST-05 error surfacing — map CCXT exceptions to api_keys.sync_status values.
    Table (per D-07):
      AuthenticationError / PermissionDenied → 'revoked'
      RateLimitExceeded → 'rate_limited'
      DeribitNotSupportedError / NetworkError / ExchangeNotAvailable / BadRequest / unknown → 'error'
    """
    if isinstance(exc, (ccxt.AuthenticationError, ccxt.PermissionDenied)):
        return "revoked"
    if isinstance(exc, ccxt.RateLimitExceeded):
        return "rate_limited"
    return "error"


async def _fetch_spot_rows(exchange_name: str, exchange) -> list[dict]:
    """
    Build spot allocator_holdings row stream from fetch_balance() + bulk fetch_tickers().

    f3 Path B: Deribit short-circuits BEFORE fetch_balance() is called — the Unified
    CCXT shape for derivatives-only accounts returns {'total': {}}, which would emit
    zero spot rows silently. Raise DeribitNotSupportedError instead so the allocator
    sees a human-readable deferral rather than a phantom "complete" with no holdings.

    Stablecoin optimization: USDT/USDC/BUSD/DAI/TUSD/USD get mark_price=1.0 WITHOUT a
    ticker call (matches RESEARCH Section 1 recommendation — lower API cost).
    """
    # f3 Path B — Deribit guard (BEFORE any network call).
    if getattr(exchange, "id", None) == "deribit":
        raise DeribitNotSupportedError(
            "Deribit spot ingestion not yet supported — derivatives still sync."
        )

    balance = await exchange.fetch_balance()
    totals = (balance.get("total") or {})
    non_zero = {asset: qty for asset, qty in totals.items() if qty and float(qty) > 0}
    if not non_zero:
        return []

    # Build the symbol list for non-stablecoin assets (CCXT market symbols like "BTC/USDT")
    need_tickers = [f"{asset}/USDT" for asset in non_zero if asset.upper() not in STABLECOINS]
    tickers: dict[str, dict] = {}
    if need_tickers:
        try:
            tickers = await exchange.fetch_tickers(need_tickers)
        except Exception:
            # Per-symbol fallback

            for sym in need_tickers:
                try:
                    tickers[sym] = await exchange.fetch_ticker(sym)
                except Exception:
                    tickers[sym] = {"last": 0.0}

    rows: list[dict] = []
    for asset, qty in non_zero.items():
        asset_upper = asset.upper()
        if asset_upper in STABLECOINS:
            mark_price = 1.0
        else:
            t = tickers.get(f"{asset}/USDT") or {}
            mark_price = float(t.get("last") or 0.0)
        rows.append({
            "venue": exchange_name,
            "symbol": asset,            # D-16: raw currency, no suffix
            "holding_type": "spot",
            "side": "flat",

            "cost_basis_usd": cost_basis,
            "raw_payload": _cap_raw_payload(s),
        })
    return rows


async def fetch_allocator_holdings(exchange_name: str, exchange) -> tuple[list[dict], str | None]:
    """
    D-01: both spot and derivatives per sync.

    Returns (rows, warning) where warning is None on full success and contains a
    sanitized message when one side succeeded and the other failed with a
    non-auth / non-rate-limit error (complete_with_warnings path — RESEARCH Q2).

    On auth or rate-limit failures OR DeribitNotSupportedError: re-raises — the handler

    a partial-success path (f3 Path B — the whole sync is marked 'error' so the
    allocator sees the deferral message; Phase 06.x will flip to Path A once a test
    key lands).
    """
    spot_rows: list[dict] = []
    deriv_rows: list[dict] = []
    warning: str | None = None

    try:
        spot_rows = await _fetch_spot_rows(exchange_name, exchange)
    except (ccxt.AuthenticationError, ccxt.PermissionDenied, ccxt.RateLimitExceeded, DeribitNotSupportedError):

        raise

    try:
        deriv_rows = await _fetch_derivative_rows(exchange_name, exchange)
    except (ccxt.AuthenticationError, ccxt.PermissionDenied, ccxt.RateLimitExceeded):
        raise
    except Exception as e:
        # Partial success — persist spot, surface warning via sync_status='complete_with_warnings'.
        warning = str(e)[:500]

    return (spot_rows + deriv_rows, warning)

            on_conflict="allocator_id,venue,symbol,asof",

    return len(rows)
