```tsx
"use client";

import React from "react";

export interface AllocatorSyncStatusProps {
  /** One of: idle | syncing | complete | complete_with_warnings | rate_limited | revoked | error. Unknown/null falls back to 'idle'. */
  syncStatus: string | null;
  /** DB value of api_keys.sync_error (already sanitized ≤500 chars server-side per D-07). */
  syncError: string | null;
  /** DB value of api_keys.last_sync_at; used to interpolate "Synced {relative time ago}" (D-08). */
  lastSyncAt: string | null;
  /** Exchange name for the rate_limited helper line (e.g. 'binance' → 'Binance cooldown remaining'). */
  exchange: string;
  /** Integer seconds until retry (for rate_limited pill). Floor to 0 if elapsed. Optional — omit to show 0s. */
  retryAtSeconds?: number;
  /**
   * f8: ISO timestamp of the queued job's next_attempt_at (from Plan 03 route's already_inflight response).
   * When syncStatus==='syncing' AND this timestamp is ≥30s in the future, render helper
   * "Queued — exchange cooldown, retry in {N}s" to surface per-exchange circuit-breaker contagion
   * from strategy-side 429s.
   */
  queuedNextAttemptAt?: string | null;
  /**
   * f4: explicit helper-text override for the manager's first-run POST failure path. When present
   * and non-empty, takes precedence over computed helper text. Supports "Sync request failed — click
   * Sync now to retry" surfacing when handleAddKey's awaited POST returns non-2xx.
   */
  helperOverride?: string | null;
}

// D-08 LOCKED VERBATIM — do NOT reword or reorder. The test suite asserts character-for-character.
// NOTE: Syncing… uses the U+2026 ellipsis (…), NOT three dots (...).
// NOTE: "Rate limited — retry in {N}s" uses the U+2014 em-dash (—), NOT hyphen-minus (-).
const PILL_STYLES: Record<string, { bg: string; text: string }> = {
  idle:                   { bg: "bg-[#F1F5F9]",   text: "text-text-secondary" },
  syncing:                { bg: "bg-[#F1F5F9]",   text: "text-text-secondary" },
  complete:               { bg: "bg-[#F1F5F9]",   text: "text-text-secondary" },
  complete_with_warnings: { bg: "bg-warning/10",  text: "text-warning" },
  rate_limited:           { bg: "bg-warning/10",  text: "text-warning" },
  revoked:                { bg: "bg-negative/10", text: "text-negative" },
  error:                  { bg: "bg-negative/10", text: "text-negative" },
};

const REVOKED_HELPER = "Re-add a read-only key from your exchange.";
const ELLIPSIS = "\u2026"; // U+2026
const EM_DASH = "\u2014";  // U+2014
const QUEUED_THRESHOLD_SECONDS = 30; // f8 — only surface "Queued" helper when breaker cooldown is ≥30s out

function formatRelative(iso: string | null): string {
  if (!iso) return "just now";
  const ts = new Date(iso).getTime();
  if (Number.isNaN(ts)) return "just now";
  const diffMs = Date.now() - ts;
  const mins = Math.floor(diffMs / 60_000);
  if (mins < 1) return "just now";
  if (mins < 60) return `${mins}m ago`;
  const hours = Math.floor(mins / 60);
  if (hours < 24) return `${hours}h ago`;
  const days = Math.floor(hours / 24);
  return `${days}d ago`;
}

function titleCase(s: string): string {
  if (!s) return s;
  return s.charAt(0).toUpperCase() + s.slice(1);
}

function secondsUntil(iso: string | null | undefined): number | null {
  if (!iso) return null;
  const target = new Date(iso).getTime();
  if (Number.isNaN(target)) return null;
  return Math.max(0, Math.floor((target - Date.now()) / 1000));
}

function SpinnerSvg() {
  return (
    <svg
      aria-hidden="true"
      width="12"
      height="12"
      viewBox="0 0 12 12"
      fill="none"
      className="motion-safe:animate-spin"
      style={{ animationDuration: "1s" }}
    >
      <circle cx="6" cy="6" r="4.5" stroke="currentColor" strokeWidth="1.5" strokeOpacity="0.25" />
      <path d="M6 1.5 A 4.5 4.5 0 0 1 10.5 6" stroke="currentColor" strokeWidth="1.5" strokeLinecap="round" />
    </svg>
  );
}

export function AllocatorSyncStatus({
  syncStatus,
  syncError,
  lastSyncAt,
  exchange,
  retryAtSeconds,
  queuedNextAttemptAt,
  helperOverride,
}: AllocatorSyncStatusProps) {
  // Forward-compat fallback: unknown / null / 'computing' → neutral idle pill per UI-SPEC.
  const normalized = (syncStatus ?? "idle") in PILL_STYLES ? (syncStatus as keyof typeof PILL_STYLES) : "idle";
  const styles = PILL_STYLES[normalized as string];

  let pillLabel: React.ReactNode;
  switch (normalized) {
    case "idle": pillLabel = "Idle"; break;
    case "syncing":
      pillLabel = (
        <span className="inline-flex items-center gap-1">
          <SpinnerSvg />
          <span>{`Syncing${ELLIPSIS}`}</span>
        </span>
      );
      break;
    case "complete":
      pillLabel = (
        <>
          Synced <span className="font-metric tabular-nums">{formatRelative(lastSyncAt)}</span>
        </>
      );
      break;
    case "complete_with_warnings": pillLabel = "Synced (warnings)"; break;
    case "rate_limited":
      pillLabel = (
        <>
          {`Rate limited ${EM_DASH} retry in `}
          <span className="font-metric tabular-nums">{Math.max(0, retryAtSeconds ?? 0)}s</span>
        </>
      );
      break;
    case "revoked": pillLabel = "Key revoked"; break;
    case "error": pillLabel = "Sync failed"; break;
  }

  // f4 override takes precedence over everything.
  let helperText = "";
  if (helperOverride && helperOverride.trim().length > 0) {
    helperText = helperOverride;
  } else if (normalized === "revoked") {
    helperText = REVOKED_HELPER;
  } else if (normalized === "rate_limited") {
    helperText = `${titleCase(exchange)} cooldown remaining`;
  } else if (normalized === "error" || normalized === "complete_with_warnings") {
    helperText = syncError ?? "";
  } else if (normalized === "syncing") {
    // f8: surface per-exchange breaker contagion when next_attempt_at ≥30s out.
    const secs = secondsUntil(queuedNextAttemptAt);
    if (secs !== null && secs >= QUEUED_THRESHOLD_SECONDS) {
      helperText = `Queued ${EM_DASH} exchange cooldown, retry in ${secs}s`;
    }
  }

  return (
    <div className="flex flex-col items-end">
      <span
        data-testid="allocator-sync-pill"
        data-sync-status={normalized}
        className={`inline-flex items-center rounded-md px-2 py-0.5 text-xs font-medium transition-colors duration-150 ease-out ${styles.bg} ${styles.text}`}
      >
        {pillLabel}
      </span>
      <div
        role="status"
        aria-live="polite"
        data-testid="allocator-sync-helper"
        className="text-xs text-text-muted mt-1"
      >
        {helperText && <span>{helperText}</span>}
      </div>
    </div>
  );
}
```
