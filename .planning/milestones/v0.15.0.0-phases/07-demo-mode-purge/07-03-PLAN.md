```typescript
import type { DailyPoint } from "@/lib/scenario";

/**
 * Convert per-allocator equity snapshots into DailyPoint[] consumed by chart widgets.
 *
 * Phase 07 / VOICES-ACCEPTED f7. Mid-series gaps are forward-filled with the previous
 * day's value_usd (naive but safe for Phase 07 MVP). A TODO to revisit for gap-aware
 * charting (e.g. explicit "no-data" markers) can be added in a later phase.
 */
export function equitySnapshotsToDailyPoints(
  snapshots: Array<{ asof: string; value_usd: number }>,
): DailyPoint[] {
  if (snapshots.length === 0) return [];
  // Defensive sort (ascending)
  const sorted = [...snapshots].sort((a, b) => a.asof.localeCompare(b.asof));
  const points: DailyPoint[] = [];
  let prevValue: number | null = null;
  let prevDate: Date | null = null;
  for (const snap of sorted) {
    const curDate = new Date(snap.asof + "T00:00:00.000Z");
    // Forward-fill any missing days between prevDate + 1 and curDate - 1
    if (prevDate !== null && prevValue !== null) {
      const oneDayMs = 24 * 60 * 60 * 1000;
      for (
        let fillDate = new Date(prevDate.getTime() + oneDayMs);
        fillDate.getTime() < curDate.getTime();
        fillDate = new Date(fillDate.getTime() + oneDayMs)
      ) {
        points.push({ date: fillDate.toISOString().slice(0, 10), value: prevValue });
      }
    }
    points.push({ date: snap.asof, value: snap.value_usd });
    prevDate = curDate;
    prevValue = snap.value_usd;
  }
  return points;
}
```
