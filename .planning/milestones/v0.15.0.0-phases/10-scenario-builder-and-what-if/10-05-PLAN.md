    ```typescript
    /**
     * Pure TypeScript — no fetch, no side effects.
     * Phase 10 D-08 + RESEARCH Pitfall 7. Computes an APPROXIMATE mandate-fit
     * tier client-side from allocator mandate preferences + strategy attributes.
     * The authoritative mandate_fit_score is engine-computed against a specific
     * allocator's universe at score time (match_candidates.score_breakdown JSONB)
     * — that source is unavailable here. This pill is INFORMATIONAL only (D-08);
     * allocators are never blocked from adding any verified strategy.
     */

    export type MandateFitTier = "green" | "yellow" | "red";

    export interface BrowseStrategyForFit {
      id: string;
      markets: string[];
      strategy_types: string[];
    }

    export interface AllocatorMandateForFit {
      preferred_markets?: string[] | null;
      excluded_strategy_types?: string[] | null;
      // Other fields available but not used by the v0.15 approximation:
      max_weight?: number | null;
      min_aum_tier?: string | null;
    }

    /**
     * Tier rubric (L2 — pinned to D-08 thresholds 0.7/0.4 verbatim for grep-consistency):
     * - HARD-RED: any strategy_type matches an excluded type → red
     * - GREEN: fraction-overlap of strategy.markets ∩ preferred_markets ≥ 0.7
     * - YELLOW: 0.4 ≤ fraction-overlap < 0.7 OR mandate missing/empty (informational fallback)
     * - RED: fraction-overlap < 0.4 (including zero overlap when mandate has preferences)
     */
    export function computeMandateFitApprox(
      strategy: BrowseStrategyForFit,
      mandate: AllocatorMandateForFit | null | undefined,
    ): MandateFitTier {
      if (!mandate) return "yellow"; // T5 — informational fallback

      const excludedTypes = new Set(mandate.excluded_strategy_types ?? []);
      for (const t of strategy.strategy_types) {
        if (excludedTypes.has(t)) return "red"; // T4 hard-red
      }

      const prefs = mandate.preferred_markets;
      if (!prefs || prefs.length === 0) return "yellow"; // T5 fallback (no signal)

      const stratMarkets = strategy.markets ?? [];
      if (stratMarkets.length === 0) return "red"; // T6 — no overlap possible

      const prefSet = new Set(prefs);
      const overlap = stratMarkets.filter((m) => prefSet.has(m)).length;
      const fraction = overlap / stratMarkets.length;

      // L2 — pinned to D-08 thresholds (0.7 / 0.4) for grep-consistency.
      if (fraction >= 0.7) return "green"; // T1
      if (fraction >= 0.4) return "yellow"; // T3 — partial fit
      return "red";                          // T2 — minimal overlap (incl. fraction=0)
    }
    ```

    ```typescript
              <button type="button" onClick={onClose} aria-label="Close drawer" className="text-text-muted hover:text-text-primary">×</button>
            </div>

              value={search}

              className="mt-4 w-full rounded-md border border-border px-3 py-2 text-sm focus:border-accent focus:outline-none focus:ring-2 focus:ring-accent/40"
    ```
