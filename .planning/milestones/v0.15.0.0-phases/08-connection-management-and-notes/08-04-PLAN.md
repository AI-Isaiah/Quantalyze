    import { Fragment, useState } from "react";

    // Note glyph — inline SVG per UI-SPEC §3
    function NoteIconSvg({ solid }: { solid: boolean }) {
      // 16×16 viewBox, document-with-lines glyph. Outlined when solid=false.
      // Implementation: use a stroke-based outlined glyph and a fill-based solid
      // glyph inside the same path; render conditionally.
      return (
        <svg width="16" height="16" viewBox="0 0 16 16" fill="none"
             stroke="currentColor" strokeWidth={solid ? 0 : 1.5} aria-hidden="true">
          {/* planner: implement the exact path strokes per UI-SPEC §3 — document outline + 3 horizontal lines */}
          <rect x="2.5" y="2" width="11" height="12" rx="1.5"
                fill={solid ? "currentColor" : "none"}
                stroke={solid ? "none" : "currentColor"} />
          <line x1="5" y1="6" x2="11" y2="6" stroke={solid ? "white" : "currentColor"} />
          <line x1="5" y1="8.5" x2="11" y2="8.5" stroke={solid ? "white" : "currentColor"} />
          <line x1="5" y1="11" x2="9" y2="11" stroke={solid ? "white" : "currentColor"} />
        </svg>
      );
    }

    export interface HoldingNoteIconButtonProps {
      hasNote: boolean;
      revoked: boolean;
      isExpanded: boolean;
      onClick: () => void;
      symbol: string;
      holdingType: "spot" | "derivative";
      rowId: string;
    }

    export function HoldingNoteIconButton(props: HoldingNoteIconButtonProps) {
      const colorClass = props.revoked
        ? "text-[#D97706]"
        : (props.hasNote ? "text-accent" : "text-text-muted");
      return (
        <button
          type="button"
          aria-label={props.hasNote
            ? `Edit note for ${props.symbol} ${props.holdingType}`
            : `Add note for ${props.symbol} ${props.holdingType}`}
          aria-expanded={props.isExpanded}
          aria-controls={`note-row-${props.rowId}`}
          onClick={props.onClick}
          className={`flex items-center justify-center h-8 w-8 rounded transition-colors hover:bg-border/50 focus-visible:ring-2 focus-visible:ring-accent/40 ${colorClass}`}
        >
          <NoteIconSvg solid={props.hasNote} />
        </button>
      );
    }

    export interface HoldingNoteRowProps {
      rowId: string;
      colSpan: number;
      venue: string;
      symbol: string;
      holding_type: "spot" | "derivative";
      initialContent: string;
      initialLastSavedAt: Date | null;
    }

    export function HoldingNoteRow(props: HoldingNoteRowProps) {
      const scope_ref = buildHoldingScopeRef({
        venue: props.venue,
        symbol: props.symbol,
        holding_type: props.holding_type,
      });
      const { saveState, lastSavedAt, save } = useNoteAutoSave("holding", scope_ref, props.initialLastSavedAt);
      const [editing, setEditing] = useState(!props.initialContent);
      const [draft, setDraft] = useState(props.initialContent);
      const [content, setContent] = useState(props.initialContent);

      return (
        <tr
          id={`note-row-${props.rowId}`}
          role="region"
          aria-label={`Note for ${props.symbol} ${props.holding_type}`}
        >
          <td colSpan={props.colSpan} className="p-0">
            <div className="px-4 py-3 bg-surface border-t border-border">
              {editing ? (
                <textarea
                  value={draft}
                  onChange={(e) => setDraft(e.target.value)}
                  onBlur={() => {
                    void save(draft);
                    setContent(draft);
                    if (draft) setEditing(false);
                  }}
                  className="w-full resize-none rounded border border-border p-2 font-mono text-[13px] leading-[1.6] focus:border-accent focus:outline-none"
                  placeholder="No note yet. Start typing to add one."
                  rows={4}
                  autoFocus
                />
              ) : (
                <div>
                  {content ? (
                    <NoteRender content={content} />
                  ) : (
                    <p className="text-sm text-text-muted">No note yet. Start typing to add one.</p>
                  )}
                  <button
                    type="button"
                    onClick={() => setEditing(true)}
                    className="mt-2 text-xs text-accent underline hover:text-accent-hover"
                  >
                    Edit
                  </button>
                </div>
              )}
              <NoteSaveStatus saveState={saveState} lastSavedAt={lastSavedAt} />
            </div>
          </td>
        </tr>
      );
    }

    import { useEffect, useState } from "react";

    export function BridgeOutcomeNoteSection({ outcomeId }: { outcomeId: string }) {
      const [content, setContent] = useState("");
      const [initialLoaded, setInitialLoaded] = useState(false);

              const json = await res.json();
              setContent(json.content ?? "");
              setDraft(json.content ?? "");
              setLastSaved(json.updated_at ? new Date(json.updated_at) : null);
            }
          } finally {
            if (!cancelled) setInitialLoaded(true);
          }
        })();
        return () => { cancelled = true; };
      }, [outcomeId]);

            <textarea
              value={draft}
              onChange={(e) => setDraft(e.target.value)}
              onBlur={() => { void save(draft); setContent(draft); if (draft) setEditing(false); }}

              <button type="button" onClick={() => setEditing(true)}
                className="mt-2 text-xs text-accent underline hover:text-accent-hover">
                Edit
              </button>
            </div>
          )}
          <NoteSaveStatus saveState={saveState} lastSavedAt={lastSavedAt} />
        </div>
