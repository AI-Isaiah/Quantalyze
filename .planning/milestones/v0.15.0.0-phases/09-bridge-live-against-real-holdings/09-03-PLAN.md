    import { NextResponse } from "next/server";
    import { z } from "zod";
    import { withAuth } from "@/lib/api/withAuth";
    import { createClient } from "@/lib/supabase/server";
    import { logAuditEvent } from "@/lib/audit";

    const BodySchema = z.object({
      holding_ref: z.string().regex(/^holding:[A-Za-z0-9_-]+:[A-Za-z0-9_-]+:[A-Za-z0-9_-]+$/, "malformed holding_ref"),
      top_candidate_strategy_id: z.string().uuid(),
    });

    function parseHoldingRef(ref: string): { venue: string; symbol: string; holding_type: string } | null {
      if (!ref.startsWith("holding:")) return null;
      const parts = ref.slice("holding:".length).split(":");
      if (parts.length !== 3) return null;
      const [venue, symbol, holding_type] = parts;
      if (!venue || !symbol || !holding_type) return null;
      return { venue, symbol, holding_type };
    }

    export const POST = withAuth(async (req: NextRequest, user): Promise<NextResponse> => {
      const supabase = await createClient();

      let body: unknown;
      try { body = await req.json(); } catch { return NextResponse.json({ error: "invalid json" }, { status: 400 }); }

      const parsed = BodySchema.safeParse(body);
      if (!parsed.success) {
        return NextResponse.json({ error: "invalid payload", issues: parsed.error.issues }, { status: 400 });
      }
      const { holding_ref, top_candidate_strategy_id } = parsed.data;

      const scope = parseHoldingRef(holding_ref);
      if (!scope) return NextResponse.json({ error: "invalid holding_ref" }, { status: 400 });

      // Ownership gate (T-09-03.b)
      const { data: ownedHolding } = await supabase
        .from("allocator_holdings")
        .select("id")
        .eq("allocator_id", user.id)
        .eq("venue", scope.venue)
        .eq("symbol", scope.symbol)
        .eq("holding_type", scope.holding_type)
        .limit(1)
        .maybeSingle();
      if (!ownedHolding) {
        return NextResponse.json({ error: "Unauthorized" }, { status: 403 });
      }

      // Strategy existence gate
      const { data: candidate } = await supabase
        .from("strategies")
        .select("id")
        .eq("id", top_candidate_strategy_id)
        .eq("status", "published")
        .maybeSingle();
      if (!candidate) {
        return NextResponse.json({ error: "strategy not found" }, { status: 404 });
      }

      // Insert match_decisions row (XOR satisfied: original_strategy_id=NULL, original_holding_ref set)
      const { data: inserted, error: insertErr } = await supabase
        .from("match_decisions")
        .insert({
          allocator_id: user.id,
          strategy_id: top_candidate_strategy_id,
          candidate_id: null,
          original_strategy_id: null,
          original_holding_ref: holding_ref,
          decision: "sent_as_intro",
          decided_by: user.id,
        })
        .select("id")
        .single();

      if (insertErr || !inserted) {
        return NextResponse.json({ error: "failed to record decision" }, { status: 500 });
      }

      // Reuse existing match.decision_record kind per D-14 (no new taxonomy)
      await logAuditEvent({
        event_type: "match.decision_record",
        actor_id: user.id,
        entity_type: "match_decision",
        entity_id: inserted.id,
        metadata: {
          original_holding_ref: holding_ref,
          top_candidate_strategy_id,
          source: "holding",
        },
      });

      return NextResponse.json({ match_decision_id: inserted.id }, { status: 201 });
    });
