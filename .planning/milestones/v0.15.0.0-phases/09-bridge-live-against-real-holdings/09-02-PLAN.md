    def reconstruct_symbol_returns(snapshots: list[dict], symbol: str) -> pd.Series | None:
        """
        Phase 09 / D-01 + D-02. Reconstruct a per-symbol daily-return Series
        from allocator_equity_snapshots rows.

        Algorithm:
        - Extract (asof, breakdown.get(symbol)) from each snapshot (ordered ASC by asof).
        - Drop entries where the symbol is absent OR zero (matches migration 073's
          extract_symbol_value_at NULLIF + (breakdown IS NULL OR value = 0) semantics).
          Rationale per RESEARCH Pitfall 2: partial days are treated as missing, NOT
          forward-filled — a deposit arriving on day N means the symbol's series
          legitimately starts at day N.
        - pct_change().dropna() over the remaining values.
        - Return None if the resulting series has < 2 points (cannot compute any return).

        Callers apply a further >= 30-day warm-up gate on the returned series per
        Phase 07 D-03 analog.
        """
        pairs: list[tuple[str, float]] = []
        for snap in snapshots:
            bd = snap.get("breakdown") or {}
            raw = bd.get(symbol)
            if raw is None:
                continue
            try:
                val = float(raw)
            except (TypeError, ValueError):
                continue
            if val == 0:
                continue
            pairs.append((snap["asof"], val))

        if len(pairs) < 2:
            return None

        asofs = [p[0] for p in pairs]
        values = [p[1] for p in pairs]
        series = pd.Series(values, index=asofs, name=symbol)
        returns = series.pct_change().dropna()
        if len(returns) == 0:
            return None
        return returns
    ```

                )

                    breaches.append("correlation_ceiling")

            top_composite: float | None = None
            slot_candidates = scored_candidates_by_slot.get(pseudo_id) or []
            # Highest final_score first; filter pseudo-ids (only real strategy UUIDs)
            for cand in sorted(slot_candidates, key=lambda c: getattr(c, "final_score", 0.0), reverse=True):
                cand_id = getattr(cand, "strategy_id", None)
                if cand_id and not str(cand_id).startswith("holding:"):
                    top_id = str(cand_id)
                    top_composite = float(getattr(cand, "final_score", 0.0))
                    break

            flagged = bool(breaches) and top_composite is not None and top_composite >= FLAG_COMPOSITE_THRESHOLD

            flags.append({
                "holding_ref": pseudo_id,
                "value_usd": value,
                "weight": weight,
                "breach_reasons": breaches,
                "top_candidate_strategy_id": top_id,
                "top_candidate_composite": top_composite,
                "flagged": flagged,
            })
        return flags
    ```
