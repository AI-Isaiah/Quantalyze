1. Using a dedicated test Supabase project (NOT preview branches — too costly and lacks seed scripts).

Per RISK-2 review feedback: the gated full-flow spec doesn't run on fork PRs (GitHub Actions doesn't pass repo secrets to forks). To maintain regression coverage on forks, an always-on smoke spec asserts the WidgetState primitive renders correctly against a placeholder build — this provides minimal but non-zero coverage on the WidgetState core when secrets are unavailable.

    /**
     * Phase 11 / D-15 — Deterministic seed for the onboarding-funnel E2E.
     *
     * Uses the test Supabase project's service-role JWT to create a fresh
     * allocator and a placeholder Bridge candidate strategy. Each spec
     * run gets its own user (timestamped email) so the suite is rerun-safe.
     *
     * Required env (assert at module load):
     *   - TEST_SUPABASE_URL
     *   - TEST_SUPABASE_SERVICE_ROLE_KEY
     */
    import { createClient, type SupabaseClient } from "@supabase/supabase-js";

    const URL_VAR = process.env.TEST_SUPABASE_URL;
    const KEY_VAR = process.env.TEST_SUPABASE_SERVICE_ROLE_KEY;

    function getAdmin(): SupabaseClient {
      if (!URL_VAR || !KEY_VAR) {
        throw new Error(
          "[seed-test-project] TEST_SUPABASE_URL or TEST_SUPABASE_SERVICE_ROLE_KEY missing — " +
          "spec must skip when secrets absent (D-16 gate / BLOCK-3 vars.E2E_TEST_DB_CONFIGURED).",
        );
      }
      return createClient(URL_VAR, KEY_VAR, {
        auth: { autoRefreshToken: false, persistSession: false },
      });
    }

    export interface SeededAllocator {
      userId: string;
      email: string;
      password: string;
    }

    export async function seedTestAllocator(): Promise<SeededAllocator> {
      const admin = getAdmin();
      const email = `e2e-onboarding-${Date.now()}@example.com`;
      const password = `e2e-${Date.now()}-${Math.random().toString(36).slice(2, 10)}`;
      const { data, error } = await admin.auth.admin.createUser({
        email,
        password,
        email_confirm: true,
      });
      if (error || !data.user) {
        throw new Error(`seedTestAllocator failed: ${error?.message ?? "no user"}`);
      }
      return { userId: data.user.id, email, password };
    }

    export interface SeededStrategy {
      strategyId: string;
    }

    export async function seedBridgeCandidate(): Promise<SeededStrategy> {
      const admin = getAdmin();
      // Minimum-viable verified strategy that the Scenario tab can pick up
      // as a Bridge candidate. Adjust columns based on the strategies table
      // shape (the executor reads supabase/migrations/001_initial_schema.sql
      // or current schema).
      const { data, error } = await admin
        .from("strategies")
        .insert({
          name: "E2E Bridge Candidate",
          codename: `e2e-bridge-${Date.now()}`,
          status: "published",
          // ... required fields per current strategies schema ...
        })
        .select("id")
        .single();
      if (error || !data) {
        throw new Error(`seedBridgeCandidate failed: ${error?.message}`);
      }
      return { strategyId: data.id };
    }

    /**
     * Phase 11 / D-15 — Teardown for the onboarding-funnel E2E.
     *
     * Cascades user deletion via auth.admin.deleteUser, which removes

    import { createClient, type SupabaseClient } from "@supabase/supabase-js";

    function getAdmin(): SupabaseClient {
      const url = process.env.TEST_SUPABASE_URL;
      const key = process.env.TEST_SUPABASE_SERVICE_ROLE_KEY;
      if (!url || !key) {
        throw new Error("[cleanup-test-project] secrets missing");

      });

    export async function cleanupTestAllocator(userId: string): Promise<void> {

      const { error } = await admin.auth.admin.deleteUser(userId);
      if (error) {

    }
