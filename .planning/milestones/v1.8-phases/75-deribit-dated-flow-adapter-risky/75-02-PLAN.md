  def deribit_dated_external_flows_usd(rows, *, supplemental_index=None) -> list[ExternalFlow]:
      by_day = defaultdict(float)
      for row in rows:
          if type not in _EXTERNAL_FLOW_TYPES: continue
          change = float(row["change"] or 0.0); if change == 0.0: continue
          day = _row_utc_day(row["timestamp"]); ccy = row["currency"].upper()
          fb = supplemental_index.get((day, ccy)) if supplemental_index else None
          by_day[day] += txn_change_to_usd(row, fallback_index=fb)
      return [ExternalFlow(day, usd) for day, usd in sorted(by_day.items())]
