```typescript
export function alignConstituentReturns(
  strategies: StrategyForBuilder[], state: ScenarioState,
): { ids: string[]; commonDates: string[]; returnsById: Record<string, number[]> };

export interface DiversificationInput {
  ids: string[];
  returnsById: Record<string, number[]>;      // ALIGNED (from alignConstituentReturns)
  weights: Record<string, number>;            // normalized, sum→1 over active
  portfolioDailyReturns: number[];            // scenarioMetrics.portfolio_daily_returns values
  correlationMatrix: Record<string, Record<string, number>> | null;
  n: number;
}
export interface DiversificationResult {
  diversificationRatio: number | null;
  effectiveNumberOfBets: number | null;
  pcr: Record<string, number> | null;        // signed, sums to 1
  clusterOrderIds: string[];                  // identity when n≤2 ids
  tooSimilarPairs: Array<[string, string, number]>;
  vols: Record<string, number> | null;
}
export function computeDiversification(input: DiversificationInput): DiversificationResult;
```
